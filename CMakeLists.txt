cmake_minimum_required(VERSION 3.21)
project(threshold_wine_xr LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(fence_wait_test tests/in_process/fence_wait_test.cpp)
add_test(NAME fence_wait COMMAND fence_wait_test)
add_executable(swapchain_wait_test tests/in_process/swapchain_wait_test.cpp)
add_test(NAME swapchain_wait COMMAND swapchain_wait_test)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_program(ZSH_EXECUTABLE zsh)
if(ZSH_EXECUTABLE)
  add_test(NAME launch_helpers COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s tests/launch -v)
  set_tests_properties(launch_helpers PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
if(APPLE)
  enable_language(OBJC)
  add_executable(native_texture_query_test tests/native/texture_query.m src/in_process/graphics_interop_native.m)
  target_include_directories(native_texture_query_test PRIVATE src/in_process)
  target_link_libraries(native_texture_query_test PRIVATE "-framework Metal" "-framework Foundation")
  add_test(NAME native_texture_query COMMAND native_texture_query_test)
  set_tests_properties(native_texture_query PROPERTIES SKIP_RETURN_CODE 77)
endif()
